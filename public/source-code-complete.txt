// ===== UTILITY FUNKCE =====

function generateUUID() {
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
    const r = Math.random() * 16 | 0;
    const v = c === 'x' ? r : (r & 0x3 | 0x8);
    return v.toString(16);
  });
}

function formatCurrency(amount) {
  return new Intl.NumberFormat('cs-CZ', {
    style: 'currency',
    currency: 'CZK',
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  }).format(amount);
}

function formatMonth(monthStr) {
  const [year, month] = monthStr.split('-');
  const date = new Date(parseInt(year), parseInt(month) - 1);
  return date.toLocaleDateString('cs-CZ', { year: 'numeric', month: 'long' });
}

function getCurrentMonth() {
  const now = new Date();
  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
}

function getCategoryPercentage(categoryAmount, totalIncome) {
  if (totalIncome === 0) return 0;
  return (categoryAmount / totalIncome) * 100;
}

function groupTransactionsByMonth(transactions) {
  const grouped = transactions.reduce((acc, transaction) => {
    const month = transaction.month;
    if (!acc[month]) {
      acc[month] = {
        month,
        income: [],
        expenses: [],
        transfers: [],
        totalIncome: 0,
        totalExpenses: 0,
        totalTransfers: 0,
        balance: 0,
        categoryBreakdown: {
          necessities: 0,
          whims: 0,
          investments: 0,
          savings: 0
        }
      };
    }

    if (transaction.type === 'income') {
      acc[month].income.push(transaction);
      acc[month].totalIncome += transaction.amount;
    } else if (transaction.type === 'expense') {
      acc[month].expenses.push(transaction);
      acc[month].totalExpenses += transaction.amount;
      if (transaction.category) {
        acc[month].categoryBreakdown[transaction.category] += transaction.amount;
      }
    } else if (transaction.type === 'transfer') {
      acc[month].transfers.push(transaction);
      acc[month].totalTransfers += transaction.amount;
    }

    return acc;
  }, {});

  Object.values(grouped).forEach((monthData) => {
    monthData.balance = monthData.totalIncome + monthData.totalTransfers - monthData.totalExpenses;
  });

  return Object.values(grouped).sort((a, b) => b.month.localeCompare(a.month));
}

// ===== EXPORT FUNKCE =====

function exportToCSV(transactions, filename = 'transactions.csv') {
  const headers = ['Datum', 'Typ', 'Název', 'Částka', 'Účet/Kategorie'];
  const rows = transactions.map((t) => [
    t.month,
    t.type === 'income' ? 'Příjem' : t.type === 'expense' ? 'Výdaj' : 'Převod',
    t.name,
    t.amount.toFixed(2),
    t.account || t.category || t.transferCategory || ''
  ]);

  const csvContent = [
    headers.join(','),
    ...rows.map((row) => row.map((cell) => `"${cell}"`).join(','))
  ].join('\n');

  const blob = new Blob(['\ufeff' + csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  link.click();
}

function exportToXLSX(transactions, filename = 'transactions.xlsx') {
  const headers = ['Datum', 'Typ', 'Název', 'Částka', 'Účet/Kategorie'];
  const rows = transactions.map((t) => [
    t.month,
    t.type === 'income' ? 'Příjem' : t.type === 'expense' ? 'Výdaj' : 'Převod',
    t.name,
    t.amount.toFixed(2),
    t.account || t.category || t.transferCategory || ''
  ]);

  const htmlTable = `
    <html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel" xmlns="http://www.w3.org/TR/REC-html40">
      <head><meta charset="utf-8" /></head>
      <body>
        <table>
          <tr>${headers.map((h) => `<th>${h}</th>`).join('')}</tr>
          ${rows.map((row) => `<tr>${row.map((cell) => `<td>${cell}</td>`).join('')}</tr>`).join('')}
        </table>
      </body>
    </html>
  `;

  const blob = new Blob([htmlTable], { type: 'application/vnd.ms-excel' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  link.click();
}

// ===== DATA MANAGEMENT =====

class FinanceApp {
  constructor() {
    this.transactions = [];
    this.bankAccounts = [];
    this.brokerAccounts = [];
    this.budgetAllocation = {
      necessities: 50,
      investments: 20,
      savings: 20,
      whims: 10
    };
    this.portfolioSettings = {
      annualReturn: 7,
      currentAge: 30
    };
    this.isDarkMode = true;
    
    this.loadFromStorage();
    this.applyTheme();
    this.render();
  }

  loadFromStorage() {
    const transactions = localStorage.getItem(STORAGE_KEYS.TRANSACTIONS);
    const bankAccounts = localStorage.getItem(STORAGE_KEYS.BANK_ACCOUNTS);
    const brokerAccounts = localStorage.getItem(STORAGE_KEYS.BROKER_ACCOUNTS);
    const budget = localStorage.getItem(STORAGE_KEYS.BUDGET);
    const portfolio = localStorage.getItem(STORAGE_KEYS.PORTFOLIO);
    const theme = localStorage.getItem(STORAGE_KEYS.THEME);

    if (transactions) this.transactions = JSON.parse(transactions);
    if (bankAccounts) this.bankAccounts = JSON.parse(bankAccounts);
    if (brokerAccounts) this.brokerAccounts = JSON.parse(brokerAccounts);
    if (budget) this.budgetAllocation = JSON.parse(budget);
    if (portfolio) this.portfolioSettings = JSON.parse(portfolio);
    if (theme) this.isDarkMode = theme === 'dark';
  }

  saveToStorage() {
    localStorage.setItem(STORAGE_KEYS.TRANSACTIONS, JSON.stringify(this.transactions));
    localStorage.setItem(STORAGE_KEYS.BANK_ACCOUNTS, JSON.stringify(this.bankAccounts));
    localStorage.setItem(STORAGE_KEYS.BROKER_ACCOUNTS, JSON.stringify(this.brokerAccounts));
    localStorage.setItem(STORAGE_KEYS.BUDGET, JSON.stringify(this.budgetAllocation));
    localStorage.setItem(STORAGE_KEYS.PORTFOLIO, JSON.stringify(this.portfolioSettings));
    localStorage.setItem(STORAGE_KEYS.THEME, this.isDarkMode ? 'dark' : 'light');
  }

  applyTheme() {
    if (this.isDarkMode) {
      document.documentElement.classList.add('dark');
    } else {
      document.documentElement.classList.remove('dark');
    }
  }

  toggleTheme() {
    this.isDarkMode = !this.isDarkMode;
    this.applyTheme();
    this.saveToStorage();
    this.render();
  }

  addTransaction(transaction) {
    const newTransaction = {
      ...transaction,
      id: generateUUID(),
      createdAt: new Date().toISOString()
    };

    this.transactions.push(newTransaction);

    // Update account balances
    if (transaction.type === 'income' && transaction.account) {
      const account = this.bankAccounts.find(acc => acc.id === transaction.account);
      if (account) {
        account.currentBalance += transaction.amount;
      }
    } else if (transaction.type === 'expense') {
      const firstAccount = this.bankAccounts[0];
      if (firstAccount) {
        firstAccount.currentBalance -= transaction.amount;
      }
    }

    localStorage.setItem(STORAGE_KEYS.LAST_TRANSACTION, JSON.stringify(transaction));
    this.saveToStorage();
    this.render();
  }

  deleteTransaction(id) {
    this.transactions = this.transactions.filter(t => t.id !== id);
    this.saveToStorage();
    this.render();
  }

  addBankAccount(name, initialBalance) {
    const newAccount = {
      id: generateUUID(),
      name,
      initialBalance,
      currentBalance: initialBalance
    };
    this.bankAccounts.push(newAccount);
    this.saveToStorage();
    this.render();
  }

  addBrokerAccount(name, initialBalance) {
    const newAccount = {
      id: generateUUID(),
      name,
      initialBalance,
      currentBalance: initialBalance
    };
    this.brokerAccounts.push(newAccount);
    this.saveToStorage();
    this.render();
  }

  getLastTransaction() {
    const last = localStorage.getItem(STORAGE_KEYS.LAST_TRANSACTION);
    return last ? JSON.parse(last) : null;
  }

  getTotalBankBalance() {
    return this.bankAccounts.reduce((sum, acc) => sum + acc.currentBalance, 0);
  }

  getTotalBrokerBalance() {
    return this.brokerAccounts.reduce((sum, acc) => sum + acc.currentBalance, 0);
  }

  // ===== RENDERING =====

  render() {
    const root = document.getElementById('root');
    root.innerHTML = `
      ${this.renderHeader()}
      <div class="flex">
        ${this.renderSidebar()}
        <main class="flex-1 p-4 md:p-8">
          ${this.renderTransactionList()}
        </main>
      </div>
    `;
    this.attachEventListeners();
  }

  renderHeader() {
    const totalBank = formatCurrency(this.getTotalBankBalance());
    const totalBroker = formatCurrency(this.getTotalBrokerBalance());
    const hasAccounts = this.bankAccounts.length > 0 || this.brokerAccounts.length > 0;

    return `
      <header class="header">
        <div class="container py-4">
          <div class="flex items-center justify-between">
            <div class="flex items-center gap-6">
              <h1 class="text-2xl font-bold text-primary">Finance App</h1>
              ${hasAccounts ? `
                <div class="flex gap-4 text-sm">
                  <div>
                    <span class="text-muted-foreground">Bankovní účty</span>
                    <div class="font-semibold text-success">${totalBank}</div>
                  </div>
                  <div>
                    <span class="text-muted-foreground">Brokerské účty</span>
                    <div class="font-semibold text-primary">${totalBroker}</div>
                  </div>
                </div>
              ` : `
                <button class="btn btn-outline btn-sm" id="btnSetupAccounts">Nastavit účty</button>
              `}
            </div>
            <div class="flex items-center gap-2">
              <button class="btn btn-primary btn-sm" id="btnNewTransaction">+ Nová transakce</button>
              <button class="btn btn-ghost btn-icon" id="btnToggleTheme">
                ${this.isDarkMode ? '☀️' : '🌙'}
              </button>
            </div>
          </div>
        </div>
      </header>
    `;
  }

  renderSidebar() {
    return `
      <aside class="sidebar">
        <nav class="space-y-2">
          <button class="btn btn-ghost w-full" style="justify-content: flex-start;">🏠 Přehled</button>
          <button class="btn btn-ghost w-full" id="btnReports" style="justify-content: flex-start;">📊 Reporty</button>
          <button class="btn btn-ghost w-full" id="btnCharts" style="justify-content: flex-start;">📈 Grafy</button>
          <div class="border-t border-border mt-4 pt-4">
            <p class="text-xs text-muted-foreground mb-2 px-3">Export dat</p>
            <button class="btn btn-ghost btn-sm w-full" id="btnExportCSV" style="justify-content: flex-start;">📄 Export CSV</button>
            <button class="btn btn-ghost btn-sm w-full" id="btnExportXLSX" style="justify-content: flex-start;">📄 Export XLSX</button>
          </div>
        </nav>
      </aside>
    `;
  }

  renderTransactionList() {
    if (this.transactions.length === 0) {
      return `
        <div class="text-center py-12">
          <p class="text-muted-foreground">Zatím nemáte žádné transakce.</p>
          <p class="text-sm text-muted-foreground mt-2">Klikněte na "Nová transakce" pro přidání první transakce.</p>
        </div>
      `;
    }

    const monthlyData = groupTransactionsByMonth(this.transactions);
    
    return `
      <div class="space-y-8">
        ${monthlyData.map(month => `
          <div class="space-y-4">
            <div class="flex items-center justify-between border-b-2 border-primary pb-2">
              <h2 class="text-2xl font-bold">${formatMonth(month.month)}</h2>
              <div class="text-right">

                      <p class="font-semibold">${formatCurrency(amount)}</p>

              </select>
            </div>
          </div>
          <button type="submit" class="btn btn-primary w-full">Přidat transakci</button>
        </form>
      </div>
    `;
    
    document.body.appendChild(overlay);
    
    // Type change handler
    const typeSelect = document.getElementById('transactionType');
    const incomeFields = document.getElementById('incomeFields');
    const expenseFields = document.getElementById('expenseFields');
    const transferFields = document.getElementById('transferFields');
    
    function updateFieldsVisibility() {
      const type = typeSelect.value;
      incomeFields.style.display = type === 'income' ? 'block' : 'none';
      expenseFields.style.display = type === 'expense' ? 'block' : 'none';
      transferFields.style.display = type === 'transfer' ? 'block' : 'none';
    }
    
    typeSelect.addEventListener('change', updateFieldsVisibility);
    updateFieldsVisibility();
    
    // Close handler
    document.getElementById('closeTransactionForm').addEventListener('click', () => overlay.remove());
    overlay.addEventListener('click', (e) => {
      if (e.target === overlay) overlay.remove();
    });
    
    // Form submit
    document.getElementById('transactionForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const formData = new FormData(e.target);
      const transaction = {
        month: formData.get('month'),
        type: formData.get('type'),
        name: formData.get('name'),
        amount: parseFloat(formData.get('amount'))
      };
      
      if (transaction.type === 'income') {
        transaction.account = formData.get('account');
      } else if (transaction.type === 'expense') {
        transaction.category = formData.get('category');
      } else if (transaction.type === 'transfer') {
        transaction.transferCategory = formData.get('transferCategory');
        transaction.transferAccount = formData.get('transferAccount');
      }
      
      this.addTransaction(transaction);
      overlay.remove();
      this.showToast('Transakce byla přidána');
    });
  }

  showCharts() {
    if (this.transactions.length === 0) {
      this.showToast('Zatím nemáte žádné transakce pro zobrazení grafů.');
      return;
    }

    const monthlyData = groupTransactionsByMonth(this.transactions);
    
    // Prepare data for charts
    const barData = monthlyData.slice().reverse();
    
    // Calculate category totals
    const categoryTotals = this.transactions
      .filter(t => t.type === 'expense')
      .reduce((acc, t) => {
        if (t.category) {
          acc[t.category] = (acc[t.category] || 0) + t.amount;
        }
        return acc;
      }, {});
    
    const overlay = document.createElement('div');
    overlay.className = 'modal-overlay';
    overlay.id = 'chartsOverlay';
    overlay.innerHTML = `
      <div class="modal-content" style="width: 100%; max-width: 72rem;">
        <div class="sticky top-0 bg-card border-b border-border px-6 py-4 flex items-center justify-between rounded-t-lg">
          <h2 class="text-xl font-semibold">Grafy a vizualizace</h2>
          <button class="btn btn-ghost btn-icon" id="closeCharts">✕</button>
        </div>
        <div class="p-6 space-y-8">
          <div class="space-y-4">
            <h3 class="text-lg font-semibold">Příjmy vs Výdaje po měsících</h3>
            <div class="chart-container bg-muted rounded-lg p-4">
              <div id="barChart" style="display: flex; align-items: flex-end; height: 100%; gap: 1rem; justify-content: center;">
                ${barData.map(month => `
                  <div style="display: flex; flex-direction: column; align-items: center; gap: 0.5rem;">
                    <div style="display: flex; gap: 4px; align-items: flex-end; height: 200px;">
                      <div style="width: 20px; background: hsl(142, 71%, 45%); height: ${month.totalIncome / Math.max(...barData.map(m => Math.max(m.totalIncome, m.totalExpenses))) * 180}px; border-radius: 2px 2px 0 0;" title="Příjmy: ${formatCurrency(month.totalIncome)}"></div>
                      <div style="width: 20px; background: hsl(0, 84%, 60%); height: ${month.totalExpenses / Math.max(...barData.map(m => Math.max(m.totalIncome, m.totalExpenses))) * 180}px; border-radius: 2px 2px 0 0;" title="Výdaje: ${formatCurrency(month.totalExpenses)}"></div>
                    </div>
                    <span class="text-xs text-muted-foreground">${formatMonth(month.month).split(' ')[0]}</span>
                  </div>
                `).join('')}
              </div>
              <div class="flex justify-center gap-6 mt-4">
                <div class="flex items-center gap-2"><div style="width: 12px; height: 12px; background: hsl(142, 71%, 45%); border-radius: 2px;"></div><span class="text-sm">Příjmy</span></div>
                <div class="flex items-center gap-2"><div style="width: 12px; height: 12px; background: hsl(0, 84%, 60%); border-radius: 2px;"></div><span class="text-sm">Výdaje</span></div>
              </div>
            </div>
          </div>
          
          ${Object.keys(categoryTotals).length > 0 ? `
          <div class="space-y-4">
            <h3 class="text-lg font-semibold">Rozdělení výdajů podle kategorií</h3>
            <div class="chart-container bg-muted rounded-lg p-4">
              <div class="grid grid-cols-2 md:grid-cols-4 gap-4">
                ${Object.entries(categoryTotals).map(([category, amount]) => {
                  const total = Object.values(categoryTotals).reduce((a, b) => a + b, 0);
                  const percent = ((amount / total) * 100).toFixed(1);
                  const colors = {
                    necessities: 'hsl(217, 91%, 60%)',
                    whims: 'hsl(0, 84%, 60%)',
                    investments: 'hsl(142, 71%, 45%)',
                    savings: 'hsl(38, 92%, 50%)'
                  };
                  return `
                    <div class="text-center">
                      <div style="width: 80px; height: 80px; border-radius: 50%; background: conic-gradient(${colors[category]} ${percent}%, hsl(var(--muted)) ${percent}%); margin: 0 auto;"></div>
                      <p class="font-semibold mt-2">${CategoryLabels[category]}</p>
                      <p class="text-sm text-muted-foreground">${formatCurrency(amount)}</p>
                      <p class="text-xs text-muted-foreground">${percent}%</p>
                    </div>
                  `;
                }).join('')}
              </div>
            </div>
          </div>
          ` : ''}
          
          <div class="space-y-4">
            <h3 class="text-lg font-semibold">Trend bilance</h3>
            <div class="chart-container bg-muted rounded-lg p-4">
              <div style="display: flex; align-items: center; height: 100%; gap: 0.5rem; justify-content: center;">
                ${barData.map((month, i) => {
                  const maxBalance = Math.max(...barData.map(m => Math.abs(m.balance)));
                  const height = Math.abs(month.balance) / maxBalance * 150;
                  const isPositive = month.balance >= 0;
                  return `
                    <div style="display: flex; flex-direction: column; align-items: center; gap: 0.5rem;">
                      <div style="height: 200px; display: flex; flex-direction: column; justify-content: center;">
                        <div style="width: 40px; background: ${isPositive ? 'hsl(217, 91%, 60%)' : 'hsl(0, 84%, 60%)'}; height: ${height}px; border-radius: 4px;" title="Bilance: ${formatCurrency(month.balance)}"></div>
                      </div>
                      <span class="text-xs text-muted-foreground">${formatMonth(month.month).split(' ')[0]}</span>
                    </div>
                  `;
                }).join('')}
              </div>
            </div>
          </div>
        </div>
      </div>
    `;
    
    document.body.appendChild(overlay);
    
    const closeModal = () => overlay.remove();
    document.getElementById('closeCharts').addEventListener('click', closeModal);
    overlay.addEventListener('click', (e) => {
      if (e.target === overlay) closeModal();
    });
  }

  showToast(message) {
    const toast = document.createElement('div');
    toast.className = 'toast';
    toast.textContent = message;
    document.body.appendChild(toast);
    setTimeout(() => toast.remove(), 3000);
  }
